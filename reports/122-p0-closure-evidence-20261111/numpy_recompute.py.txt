def independent_error(prediction, truth):
    return np.abs(np.where(prediction < 0, 0.0, prediction) - np.log(np.where(truth < 1, 1.0, truth)))


def independent_metrics(prediction, truth):
    error = independent_error(prediction, truth)
    return {'n': len(error), 'log_q95': float(np.quantile(error, 0.95, method='linear')), **{f'q{p}': float(np.quantile(np.exp(error), p / 100, method='linear')) for p in (50, 90, 95, 99)}}

def paired_case(deltas):
    deltas = np.asarray(deltas, dtype=float)
    mean = float(deltas.sum() / len(deltas))
    sd = float(np.sqrt(np.sum((deltas - mean) ** 2) / (len(deltas) - 1)))
    passed = mean < 0 and -mean >= sd and (sd != 0 or np.all(deltas < 0))
    return {'mean': mean, 'sd_ddof1': sd, 'passed': bool(passed), 'per_seed': deltas.tolist()}
