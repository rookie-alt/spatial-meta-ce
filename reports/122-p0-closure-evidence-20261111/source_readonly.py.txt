def sha(path):
    value = hashlib.sha256()
    with path.open('rb') as stream:
        for chunk in iter(lambda: stream.read(4 * 1024 * 1024), b''):
            value.update(chunk)
    return value.hexdigest()


def digest(obj):
    return hashlib.sha256(json.dumps(obj, sort_keys=True, ensure_ascii=False).encode()).hexdigest()


def load(path):
    return json.loads(path.read_text())


def plan_digest(obj):
    return digest({k: v for k, v in obj.items() if k != 'plan_sha256'})


raw = load(OUTPUT / 'preflight.json')
assert raw['status'] == 'data_preflight_passed'
assert raw['p0_gate_passed'] is False and raw['formal_training_authorized'] is False
assert raw['target_databases_read'] == []

assert tuple(capacity['databases']) == TRAIN
assert digest({k: v for k, v in capacity.items() if k != 'capacity_sha256'}) == capacity['capacity_sha256']
records = {}
identities = {}
for db in TRAIN + SOURCE:
    directory = CACHE / db
    manifest_path = directory / 'manifest.json'
    manifest_sha = sha(manifest_path)
    manifest = load(manifest_path)
    assert manifest['schema'] == 1
    assert manifest['transform'] == 'price40-group-logfanout-v1'
    assert digest({k: v for k, v in manifest.items() if k != 'content_hash'}) == manifest['content_hash']
    rows = manifest['rows']
    identities[db] = tuple((row['id'], row['query_hash']) for row in rows)
    offsets = np.load(directory / 'offsets.npy', mmap_mode='r', allow_pickle=False)
    assert offsets.shape == (len(rows) + 1, 4)
    assert offsets.dtype.kind in 'iu' and np.all(offsets[0] == 0)
    counts = np.diff(offsets, axis=0)
    assert np.all(counts >= 0)
    maxima = counts.max(axis=0).tolist()
    assert maxima == manifest['inspection']['max_counts']
    unique = len({row['query_hash'] for row in rows})
    assert unique == manifest['unique_queries']
    expected = capacity if db in TRAIN else raw['coverage']['databases'][db]
    assert manifest_sha == (expected['manifest_sha256'][db] if db in TRAIN else expected['manifest_sha256'])
    assert manifest['content_hash'] == (expected['cache_hashes'][db] if db in TRAIN else expected['cache_hash'])
    if db in TRAIN:
        assert maxima == capacity['database_max_counts'][db]
    else:
        assert maxima == expected['max_counts']
    files = {name: sha(directory / name) for name in ('cards.npy', 'offsets.npy', 'group0.npy', 'group1.npy', 'group2.npy', 'group3.npy')}
    data_sha = digest({'cards': files['cards.npy'], 'offsets': files['offsets.npy'], 'groups': [files[f'group{i}.npy'] for i in range(4)]})
    assert sha(manifest_path) == manifest_sha
    records[db] = {'manifest_sha256': manifest_sha, 'content_hash': manifest['content_hash'], 'rows': len(rows), 'unique_queries': unique, 'max_counts': maxima, 'unsupported_rows': int(np.any(counts > np.asarray(capacity['max_counts']), axis=1).sum()), 'array_sha256': files, 'data_sha256': data_sha, 'upstream_provenance_recorded_not_reread': manifest['provenance']}
    del manifest, rows, offsets, counts
assert np.max(np.asarray([records[db]['max_counts'] for db in TRAIN]), axis=0).tolist() == capacity['max_counts'] == [12, 12, 7, 10]
